function applyZoom(){ var sliderLevel = parseInt(zoomSlider.value, 10) / 100; zoomValue.textContent = zoomSlider.value + '%'; var w = window.innerWidth; if (w <= 480){ // 手机端：完全不缩放 mainWrap.style.zoom = ''; mainWrap.style.transform = ''; return; } // 电脑/平板：根据屏幕宽度自动等比放大 var baseZoom = Math.min(2, Math.max(1, (w - 60) / 390)); var finalZoom = baseZoom * sliderLevel; mainWrap.style.zoom = finalZoom; } window.addEventListener('resize', applyZoom); applyZoom();
